```
"""Circuit breaker that notifies subscribers through NOTIFIER."""

import warnings

import breaker_events
from breaker_events import Notifier
from breaker_state import BreakerState
from call_context import CallContext


class CircuitBreaker:
    """Tracks failures for a named dependency and notifies NOTIFIER.

    The breaker does not inspect, filter, remap, or reinterpret
    CallContext.caller_state. It holds no references to other systems;
    NOTIFIER is its sole point of contact.
    """

    def __init__(self, name: str, failure_threshold: int = 5) -> None:
        """Initialize a breaker.

        Args:
            name: Breaker name passed to notifier events.
            failure_threshold: Number of accepted failures required to open.
        """
        self.name = name
        self.failure_threshold = failure_threshold
        self._failure_count: int = 0
        self._open: bool = False

    def record_failure(self, context: CallContext) -> None:
        """Record a failure and notify subscribers.

        Invalid contexts with negative duration_ms are dropped with a warning.
        Failures on an already-open breaker are ignored. The internal state is
        mutated before any notification, so reentrant calls see the post-mutation
        state.
        """
        if context.duration_ms < 0:
            warnings.warn(
                f"CircuitBreaker {self.name!r} dropped negative duration_ms={context.duration_ms!r}",
                RuntimeWarning,
                stacklevel=2,
            )
            return

        if self._open:
            return

        self._failure_count += 1
        failure_count = self._failure_count
        opened_now = False
        if failure_count >= self.failure_threshold:
            self._open = True
            opened_now = True

        name = self.name
        threshold = self.failure_threshold
        notifier: Notifier = breaker_events.NOTIFIER

        notifier.emit_call_failed(name, context, failure_count, threshold)
        if opened_now:
            notifier.emit_breaker_opened(name, context)

    def record_success(self, context: CallContext) -> None:
        """Record a success and notify subscribers.

        Invalid contexts with negative duration_ms are dropped with a warning.
        If the breaker is not open, the failure count is reset before the
        success notification. If the breaker is open, the count is not reset.
        """
        if context.duration_ms < 0:
            warnings.warn(
                f"CircuitBreaker {self.name!r} dropped negative duration_ms={context.duration_ms!r}",
                RuntimeWarning,
                stacklevel=2,
            )
            return

        if not self._open:
            self._failure_count = 0

        current_state = BreakerState.OPEN if self._open else BreakerState.CLOSED
        name = self.name
        notifier: Notifier = breaker_events.NOTIFIER

        notifier.emit_call_succeeded(name, context, current_state)

    def reset(self) -> None:
        """Return the breaker to CLOSED with a zero failure count.

        This method emits no notifications.
        """
        self._failure_count = 0
        self._open = False
```
